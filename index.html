-- Enable the PostGIS extension (available in Supabase by default)
CREATE EXTENSION IF NOT EXISTS postgis;

-- Create the service providers table with a Geography column
CREATE TABLE service_providers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    category TEXT NOT NULL,
    description TEXT,
    -- geography(Point, 4326) represents points on the WGS 84 ellipsoid (standard GPS coords)
    location geography(Point, 4326) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Create a GIST spatial index for high-speed location-based coordinate indexing
CREATE INDEX service_providers_location_idx ON service_providers USING gist (location);
-- Create a custom database function (RPC) to calculate range query & distance
CREATE OR REPLACE FUNCTION get_providers_within_radius(
    user_lat DOUBLE PRECISION,
    user_lng DOUBLE PRECISION,
    radius_meters DOUBLE PRECISION
)
RETURNS TABLE (
    id UUID,
    name TEXT,
    category TEXT,
    distance_meters DOUBLE PRECISION
) 
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT 
        sp.id,
        sp.name,
        sp.category,
        -- ST_Distance computes shortest distance in meters on the ellipsoid
        ST_Distance(sp.location, ST_SetSRID(ST_MakePoint(user_lng, user_lat), 4326)::geography) AS distance_meters
    FROM 
        service_providers sp
    WHERE 
        -- ST_DWithin utilizes the GIST spatial index for millisecond performance
        ST_DWithin(
            sp.location, 
            ST_SetSRID(ST_MakePoint(user_lng, user_lat), 4326)::geography, 
            radius_meters
        )
    ORDER BY 
        distance_meters;
END;
$$;

Vite + React and Supabase Integration

To connect this architecture to a real React frontend, run the following steps:

  1. Install dependencies: 
    npm install @supabase/supabase-js react-leaflet leaflet
  2. Initialize client: 
    import { createClient } from '@supabase/supabase-js';
const supabase = createClient(VITE_SUPABASE_URL, VITE_SUPABASE_ANON_KEY);
  3. Query matching React hook: 
    async function fetchNearbyProviders(lat, lng, radiusKm) {
  const { data, error } = await supabase.rpc('get_providers_within_radius', {
    user_lat: lat,
    user_lng: lng,
    radius_meters: radiusKm * 1000
  });
  if (error) console.error("Database query failed:", error);
  return data;
}

Low-Cost & Open-Source Map Architectures

For budget-conscious apps, avoid Google Maps' high volume API charges ($7.00 CPM). Use these open stack patterns instead:

Provider Stack Approx. Cost Features / Benefits
Leaflet + CartoDB Tiles (Current Demo) $0.00 / Free Lightweight, simple raster map tiles, no API keys, standard look. Great for quick MVPs.
MapLibre GL + MapTiler Free Tier (100k views/mo, then $0.50/k) GPU-accelerated vector mapping, tilt, custom styling. Elegant visual design.
Protomaps (Self-hosted PMTiles) ~$0.0001 per gigabyte (S3/Cloudflare costs) Entire planet or city vector database stored in a single file on AWS S3/Cloudflare. Zero recurring provider fees.
